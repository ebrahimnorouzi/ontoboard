robot --catalog catalog-v001.xml convert --input comp-edit.owl --format ofn --output tmp/comp-preprocess.owl
robot --catalog catalog-v001.xml merge -I https://example.org/fetched.owl \
	 annotate --annotation owl:versionInfo 1.0.0 --output tmp/component-download-fetched.owl.owl
touch tmp/stamp-component-fetched.owl
if cmp -s tmp/component-download-fetched.owl.owl tmp/component-download-fetched.owl.tmp.owl ; then \
        echo "Component identical." ; \
else \
        echo "Component different, updating." && \
        cp tmp/component-download-fetched.owl.owl tmp/component-download-fetched.owl.tmp.owl && \
        robot --catalog catalog-v001.xml annotate --input tmp/component-download-fetched.owl.owl \
                          --ontology-iri http://purl.obolibrary.org/obo/comp/components/fetched.owl annotate -V http://purl.obolibrary.org/obo/comp/releases/1.0.0/components/fetched.owl --annotation owl:versionInfo 1.0.0 \
                          --output components/fetched.owl ; \
fi
touch tmp/stamp-component-fromtemplate.owl
robot --catalog catalog-v001.xml template  \
	 --template ../templates/first.tsv \
	 --template ../templates/second.tsv \
	 annotate --ontology-iri http://purl.obolibrary.org/obo/comp/components/fromtemplate.owl annotate -V http://purl.obolibrary.org/obo/comp/releases/1.0.0/components/fromtemplate.owl --annotation owl:versionInfo 1.0.0 convert -f ofn --output components/fromtemplate.owl.tmp.owl && mv components/fromtemplate.owl.tmp.owl components/fromtemplate.owl
robot --catalog catalog-v001.xml remove --input tmp/comp-preprocess.owl --select imports --trim false \
	 merge  --input components/manual.owl  --input components/fetched.owl  --input components/fromtemplate.owl \
	       --output tmp/merged-comp-edit.ofn
mkdir -p reports
robot --catalog catalog-v001.xml verify -i tmp/merged-comp-edit.ofn --queries ../sparql/owldef-self-reference-violation.sparql ../sparql/iri-range-violation.sparql ../sparql/label-with-iri-violation.sparql ../sparql/multiple-replaced_by-violation.sparql ../sparql/dc-properties-violation.sparql -O reports
