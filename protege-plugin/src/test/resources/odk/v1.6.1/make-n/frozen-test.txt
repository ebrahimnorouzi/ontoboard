	dicer-cli policy --assume-manchester --show-owlapi-error froz-idranges.owl ; \
fi
robot --catalog catalog-v001.xml convert --input froz-edit.owl --format ofn --output tmp/froz-preprocess.owl
robot --catalog catalog-v001.xml reason --input tmp/froz-preprocess.owl --reasoner ELK --equivalent-classes-allowed asserted-only \
	--exclude-tautologies structural --output test.owl && rm test.owl
robot --catalog catalog-v001.xml remove --input tmp/froz-preprocess.owl --select imports --trim false \
	 merge  \
	       --output tmp/merged-froz-edit.ofn
mkdir -p reports
robot --catalog catalog-v001.xml verify -i tmp/merged-froz-edit.ofn --queries ../sparql/owldef-self-reference-violation.sparql ../sparql/iri-range-violation.sparql ../sparql/label-with-iri-violation.sparql ../sparql/multiple-replaced_by-violation.sparql -O reports
robot --catalog catalog-v001.xml report -i tmp/merged-froz-edit.ofn -l true --profile profile.txt --fail-on ERROR --print 5 -o reports/froz-edit.owl-obo-report.tsv
robot --catalog catalog-v001.xml convert -I https://example.org/handmade.owl -o tmp/mirror-handmade.owl
mkdir -p mirror
if [ -f tmp/mirror-handmade.owl ]; then if cmp -s tmp/mirror-handmade.owl mirror/handmade.owl ; then echo "Mirror identical, ignoring."; else echo "Mirrors different, updating." &&\
	cp tmp/mirror-handmade.owl mirror/handmade.owl; fi; fi
echo "ERROR: You have configured handmade as a custom module;"
echo "       This rule needs to be overwritten in froz.Makefile!"
false
robot --catalog catalog-v001.xml convert -I https://example.org/mini.owl -o tmp/mirror-mini.owl
if [ -f tmp/mirror-mini.owl ]; then if cmp -s tmp/mirror-mini.owl mirror/mini.owl ; then echo "Mirror identical, ignoring."; else echo "Mirrors different, updating." &&\
	cp tmp/mirror-mini.owl mirror/mini.owl; fi; fi
robot --catalog catalog-v001.xml query --input tmp/merged-froz-edit.ofn --format --csv \
	       --query ../sparql/terms.sparql tmp/pre_seed.txt
cat tmp/pre_seed.txt | sort | uniq > tmp/seed.txt
mkdir -p tmp/plugins
if [ -f ../../plugins/kgcl.jar ]; then        \
	ln ../../plugins/kgcl.jar tmp/plugins/kgcl.jar ;         \
elif [ -f /tools/robot-plugins/kgcl.jar ]; then \
	cp /tools/robot-plugins/kgcl.jar tmp/plugins/kgcl.jar ;  \

mkdir -p tmp/plugins
if [ -f ../../plugins/odk.jar ]; then        \
	ln ../../plugins/odk.jar tmp/plugins/odk.jar ;         \
elif [ -f /tools/robot-plugins/odk.jar ]; then \
	cp /tools/robot-plugins/odk.jar tmp/plugins/odk.jar ;  \
fi
mkdir -p tmp/plugins
if [ -f ../../plugins/sssom.jar ]; then        \
	ln ../../plugins/sssom.jar tmp/plugins/sssom.jar ;         \
elif [ -f /tools/robot-plugins/sssom.jar ]; then \
	cp /tools/robot-plugins/sssom.jar tmp/plugins/sssom.jar ;  \
fi
robot --catalog catalog-v001.xml annotate --input mirror/mini.owl --remove-annotations \
	 odk:normalize --add-source true \
	 extract --term-file imports/mini_terms.txt --term-file tmp/seed.txt \
	         --force true --copy-ontology-annotations true \
	         --individuals include \
	         --method BOT \
	 remove  --term rdfs:label  --term IAO:0000115  --term OMO:0002000 \
	        --term-file imports/mini_terms.txt --term-file tmp/seed.txt \
	        --select complement --select annotation-properties \
	 odk:normalize --base-iri https://example.org/froz \
	               --subset-decls true --synonym-decls true \
	 repair --merge-axiom-annotations true \
	 annotate --ontology-iri https://example.org/froz/froz/imports/mini_import.owl annotate -V https://example.org/froz/froz/releases/1.0.0/imports/mini_import.owl --annotation owl:versionInfo 1.0.0 convert -f ofn --output imports/mini_import.owl.tmp.owl && mv imports/mini_import.owl.tmp.owl imports/mini_import.owl
robot --catalog catalog-v001.xml merge --input tmp/froz-preprocess.owl  \
	reason --reasoner ELK --equivalent-classes-allowed asserted-only --exclude-tautologies structural \
	relax --include-subclass-of true \
	reduce -r ELK --include-subproperties true \
	 annotate --ontology-iri https://example.org/froz/froz/froz-full.owl annotate -V https://example.org/froz/froz/releases/1.0.0/froz-full.owl --annotation owl:versionInfo 1.0.0 --output froz-full.owl.tmp.owl && mv froz-full.owl.tmp.owl froz-full.owl
robot --catalog catalog-v001.xml annotate --input froz-full.owl --ontology-iri https://example.org/froz/froz.owl annotate -V https://example.org/froz/froz/releases/1.0.0/froz.owl --annotation owl:versionInfo 1.0.0 \
	convert -o froz.owl.tmp.owl && mv froz.owl.tmp.owl froz.owl
robot --catalog catalog-v001.xml merge -i froz.owl convert -f ofn -o tmp/validate.ofn
robot --catalog catalog-v001.xml validate-profile --profile DL -i tmp/validate.ofn -o reports/validate_profile_owl2dl_froz.owl.txt || { cat reports/validate_profile_owl2dl_froz.owl.txt && exit 1; }
