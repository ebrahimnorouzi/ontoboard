echo "ODK Makefile v1.6.1"
odk-info --tools
if [ -f obim-idranges.owl ]; then \
	dicer-cli policy --assume-manchester --show-owlapi-error obim-idranges.owl ; \
fi
robot --catalog catalog-v001.xml convert --input obim-edit.owl --format ofn --output tmp/obim-preprocess.owl
robot --catalog catalog-v001.xml reason --input tmp/obim-preprocess.owl --reasoner ELK --equivalent-classes-allowed asserted-only \
	--exclude-tautologies structural --output test.owl && rm test.owl
robot --catalog catalog-v001.xml remove --input tmp/obim-preprocess.owl --select imports --trim false \
	 merge  \
	       --output tmp/merged-obim-edit.ofn
mkdir -p reports
robot --catalog catalog-v001.xml verify -i tmp/merged-obim-edit.ofn --queries ../sparql/owldef-self-reference-violation.sparql ../sparql/iri-range-violation.sparql ../sparql/label-with-iri-violation.sparql ../sparql/multiple-replaced_by-violation.sparql ../sparql/dc-properties-violation.sparql -O reports
robot --catalog catalog-v001.xml report -i tmp/merged-obim-edit.ofn -l true  --fail-on None --base-iri http://purl.obolibrary.org/obo/OBIM_ --base-iri http://purl.obolibrary.org/obo/obim --print 5 -o reports/obim-edit.owl-obo-report.tsv
robot --catalog catalog-v001.xml merge --input tmp/obim-preprocess.owl  \
	reason --reasoner ELK --equivalent-classes-allowed asserted-only --exclude-tautologies structural \
	relax --include-subclass-of true \
	reduce -r ELK --include-subproperties true \
	 annotate --ontology-iri http://purl.obolibrary.org/obo/obim/obim-full.owl annotate -V http://purl.obolibrary.org/obo/obim/releases/1.0.0/obim-full.owl --annotation owl:versionInfo 1.0.0 --output obim-full.owl.tmp.owl && mv obim-full.owl.tmp.owl obim-full.owl
robot --catalog catalog-v001.xml annotate --input obim-full.owl --ontology-iri http://purl.obolibrary.org/obo/obim.owl annotate -V http://purl.obolibrary.org/obo/obim/releases/1.0.0/obim.owl --annotation owl:versionInfo 1.0.0 \
	convert -o obim.owl.tmp.owl && mv obim.owl.tmp.owl obim.owl
robot --catalog catalog-v001.xml merge -i obim.owl convert -f ofn -o tmp/validate.ofn
robot --catalog catalog-v001.xml validate-profile --profile DL -i tmp/validate.ofn -o reports/validate_profile_owl2dl_obim.owl.txt || { cat reports/validate_profile_owl2dl_obim.owl.txt && exit 1; }
echo "Finished running all tests successfully."
