echo "ODK Makefile v1.6.1"
odk-info --tools
if [ -f loim-idranges.owl ]; then \
	dicer-cli policy --assume-manchester --show-owlapi-error loim-idranges.owl ; \
fi
robot --catalog catalog-v001.xml convert --input loim-edit.owl --format ofn --output tmp/loim-preprocess.owl
robot --catalog catalog-v001.xml reason --input tmp/loim-preprocess.owl --reasoner ELK --equivalent-classes-allowed asserted-only \
	--exclude-tautologies structural --output test.owl && rm test.owl
robot --catalog catalog-v001.xml remove --input tmp/loim-preprocess.owl --select imports --trim false \
	 merge  \
	       --output tmp/merged-loim-edit.ofn
mkdir -p reports
robot --catalog catalog-v001.xml verify -i tmp/merged-loim-edit.ofn --queries ../sparql/owldef-self-reference-violation.sparql ../sparql/iri-range-violation.sparql ../sparql/label-with-iri-violation.sparql ../sparql/multiple-replaced_by-violation.sparql ../sparql/dc-properties-violation.sparql -O reports
robot --catalog catalog-v001.xml report -i tmp/merged-loim-edit.ofn -l true  --fail-on None --base-iri http://purl.obolibrary.org/obo/LOIM_ --base-iri http://purl.obolibrary.org/obo/loim --print 5 -o reports/loim-edit.owl-obo-report.tsv
robot --catalog catalog-v001.xml merge --input tmp/loim-preprocess.owl  \
	reason --reasoner ELK --equivalent-classes-allowed asserted-only --exclude-tautologies structural \
	relax --include-subclass-of true \
	reduce -r ELK --include-subproperties true \
	 annotate --ontology-iri http://purl.obolibrary.org/obo/loim/loim-full.owl annotate -V http://purl.obolibrary.org/obo/loim/releases/1.0.0/loim-full.owl --annotation owl:versionInfo 1.0.0 --output loim-full.owl.tmp.owl && mv loim-full.owl.tmp.owl loim-full.owl
robot --catalog catalog-v001.xml annotate --input loim-full.owl --ontology-iri http://purl.obolibrary.org/obo/loim.owl annotate -V http://purl.obolibrary.org/obo/loim/releases/1.0.0/loim.owl --annotation owl:versionInfo 1.0.0 \
	convert -o loim.owl.tmp.owl && mv loim.owl.tmp.owl loim.owl
robot --catalog catalog-v001.xml merge -i loim.owl convert -f ofn -o tmp/validate.ofn
robot --catalog catalog-v001.xml validate-profile --profile DL -i tmp/validate.ofn -o reports/validate_profile_owl2dl_loim.owl.txt || { cat reports/validate_profile_owl2dl_loim.owl.txt && exit 1; }
echo "Finished running all tests successfully."
